Code Examples

import apgi_sdk

# Initialize secure consciousness simulation
client = apgi_sdk.Client(api_key="YOUR_API_KEY")

session = client.sessions.create(
    model="apgi-ignition-v2",
    parameters={
        "threshold": 0.85,
        "precision_gating": True
    }
)

print(f"Ignition State: {session.state}")
const { APGI } = require('@apgi/sdk');

// Connect to the neural engine
const client = new APGI({ apiKey: 'YOUR_API_KEY' });

async function startSimulation() {
  const session = await client.sessions.initialize({
    architecture: 'allostatic-core',
    realtime: true
  });
  
  console.log(`Node Active: ${session.id}`);
}
